#!/bin/bash

set -e  # Exit immediately if any command fails

echo "========== Kubernetes Common Node Setup Started =========="

# ------------------------------------------------------------
# 1. Disable Swap (Kubernetes Requirement)
# ------------------------------------------------------------

echo "[STEP 1] Disabling Swap..."

swapoff -a

# Comment out swap entry in fstab (if exists)
sed -i '/\sswap\s/s/^/#/' /etc/fstab

echo "Swap disabled successfully."


# ------------------------------------------------------------
# 2. Enable Required Kernel Modules
# ------------------------------------------------------------

echo "[STEP 2] Loading Kernel Modules..."

cat <<EOF | tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF

modprobe overlay
modprobe br_netfilter

echo "Kernel modules loaded."


# ------------------------------------------------------------
# 3. Configure sysctl parameters for Kubernetes Networking
# ------------------------------------------------------------

echo "[STEP 3] Configuring sysctl..."

cat <<EOF | tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
EOF

sysctl --system

echo "Sysctl configured successfully."


# ------------------------------------------------------------
# 4. Install Containerd (Container Runtime)
# ------------------------------------------------------------

echo "[STEP 4] Installing Containerd..."

apt update -y
apt install -y containerd

mkdir -p /etc/containerd

containerd config default | tee /etc/containerd/config.toml

# Enable SystemdCgroup = true (Production Requirement)
sed -i 's/SystemdCgroup = false/SystemdCgroup = true/' /etc/containerd/config.toml

systemctl restart containerd
systemctl enable containerd

echo "Containerd installed and configured."


# ------------------------------------------------------------
# 5. Install Kubernetes Components
# ------------------------------------------------------------

echo "[STEP 5] Installing kubeadm, kubelet, kubectl..."

apt install -y apt-transport-https ca-certificates curl gpg

mkdir -p /etc/apt/keyrings

curl -fsSL https://pkgs.k8s.io/core:/stable:/v1.28/deb/Release.key \
  | gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg

echo "deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] \
https://pkgs.k8s.io/core:/stable:/v1.28/deb/ /" \
  | tee /etc/apt/sources.list.d/kubernetes.list

apt update -y

apt install -y kubelet kubeadm kubectl

apt-mark hold kubelet kubeadm kubectl

systemctl enable kubelet

echo "Kubernetes components installed successfully."


echo "========== Common Node Setup Completed Successfully =========="
